Llama-3.2-3B-Instruct railway def computeFee(basePrice, n_passengers, n_over18, n_under15):
    """
    Compute the fee for a group of passengers.

    Args:
    basePrice (float): The base price per passenger.
    n_passengers (int): The total number of passengers.
    n_over18 (int): The number of passengers aged 18 or older.
    n_under15 (int): The number of passengers under 15.

    Returns:
    float: The total fee for the group.

    Raises:
    ValueError: If the inputs are invalid.
    """

    # Check if the inputs are valid
    if not isinstance(basePrice, (int, float)) or basePrice < 0:
        raise ValueError("Invalid base price")
    if not isinstance(n_passengers, int) or n_passengers <= 0:
        raise ValueError("Invalid number of passengers")
    if not isinstance(n_over18, int) or n_over18 < 0:
        raise ValueError("Invalid number of passengers over 18")
    if not isinstance(n_under15, int) or n_under15 < 0:
        raise ValueError("Invalid number of passengers under 15")

    # Check if the counts are consistent with the group size
    if n_passengers < 2 or n_passengers > 5:
        raise ValueError("Invalid group size")
    if n_over18 + n_under15 > n_passengers:
        raise ValueError("Invalid counts")

    # Compute the fee
    fee = 0
    for i in range(2, 6):
        # Calculate the number of groups of i passengers
        groups = n_passengers // i
        # Calculate the number of passengers in each group
        passengers_in_group = n_passengers % i
        # Calculate the fee for each group
        group_fee = basePrice * i
        # Add the fee for the current group to the total fee
        fee += group_fee * groups
        # Add the fee for the remaining passengers to the total fee
        fee += basePrice * passengers_in_group

    # Subtract the fee for passengers under 15 from the total fee
    fee -= n_under15 * basePrice

    return fee

Llama-3.2-3B-Instruct bike def computeMaxTime(winner_time, avg_speed, track_type):
    """
    Compute the maximum time a winner can have on a track.

    Args:
    winner_time (float): The initial time of the winner.
    avg_speed (float): The average speed of the winner.
    track_type (str): The type of the track. Can be 'A', 'B', or 'C'.

    Returns:
    float: The maximum time the winner can have on the track.
    """
    
    # Check if inputs are valid
    if not isinstance(winner_time, (int, float)) or winner_time <= 0:
        return 0
    if not isinstance(avg_speed, (int, float)) or avg_speed <= 0:
        return 0
    if not isinstance(track_type, str) or len(track_type)!= 1 or not track_type.isupper():
        return 0

    # Define the track types and their corresponding time multipliers
    track_types = {
        'A': {'speed': 30,'multiplier': 0.05},
        'B': {'speed': 35,'multiplier': 0.20},
        'C': {'speed': float('inf'),'multiplier': 0.50}
    }

    # Check if the track type is valid
    if track_type not in track_types:
        return 0

    # Calculate the multiplier based on the speed
    if avg_speed <= track_types[track_type]['speed']:
        multiplier = track_types[track_type]['multiplier']
    else:
        multiplier = 0

    # Calculate the maximum time
    max_time = winner_time * (1 + multiplier)

    return max_time
